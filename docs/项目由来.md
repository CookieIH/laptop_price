# 自定义性能评分规则
def performance_score(row):
    score = 0
    # CPU权重0.35
    if row['cpu_brand'] == 'Intel':
        if 'i9' in row['cpu_model']: score += 35
        elif 'i7' in row['cpu_model']: score += 28
        elif 'i5' in row['cpu_model']: score += 20
    # 内存权重0.25
    score += row['memory'] / 32 * 25
    # 显卡权重0.25
    if 'RTX 40' in row['gpu_model']: score += 25
    elif 'RTX 30' in row['gpu_model']: score += 18
    # 硬盘权重0.15
    score += row['storage'] / 1024 * 15
    return score

row['performance'] = performance_score(row)
row['value_for_money'] = row['performance'] / row['price'] * 1000  # 性价比指数

from fastapi import APIRouter, Depends, Query, HTTPException
from sqlalchemy.orm import Session
from typing import Optional

from ..database import get_db
from ..models import Laptop
from ..schemas import LaptopResponse, LaptopFilterParams

router = APIRouter()

@router.get("/", response_model=list[LaptopResponse])
async def get_laptops(
    brand: Optional[str] = Query(None, description="品牌筛选"),
    min_price: Optional[float] = Query(None, description="最低价格"),
    max_price: Optional[float] = Query(None, description="最高价格"),
    min_memory: Optional[int] = Query(None, description="最小内存(GB)"),
    max_memory: Optional[int] = Query(None, description="最大内存(GB)"),
    limit: int = Query(100, ge=1, le=500, description="返回条数"),
    offset: int = Query(0, ge=0, description="偏移量"),
    db: Session = Depends(get_db)
):
    """获取笔记本列表（支持多条件筛选）"""
    query = db.query(Laptop)

    if brand:
        query = query.filter(Laptop.brand == brand)
    if min_price is not None:
        query = query.filter(Laptop.price >= min_price)
    if max_price is not None:
        query = query.filter(Laptop.price <= max_price)
    if min_memory is not None:
        query = query.filter(Laptop.memory >= min_memory)
    if max_memory is not None:
        query = query.filter(Laptop.memory <= max_memory)

    laptops = query.offset(offset).limit(limit).all()
    return laptops

@router.get("/{laptop_id}", response_model=LaptopResponse)
async def get_laptop(laptop_id: int, db: Session = Depends(get_db)):
    """获取单条笔记本详情"""
    laptop = db.query(Laptop).filter(Laptop.id == laptop_id).first()
    if not laptop:
        raise HTTPException(status_code=404, detail="笔记本不存在")
    return laptop

from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from sqlalchemy import func, and_
from datetime import datetime, timedelta

from ..database import get_db
from ..models import Laptop
from ..schemas import (
    BrandAvgResponse,
    PriceDistributionItem,
    TrendItem,
    ScatterItem,
    RatioRankItem
)

router = APIRouter()

@router.get("/brand_avg", response_model=list[BrandAvgResponse])
async def get_brand_avg(db: Session = Depends(get_db)):
    """各品牌均价统计"""
    results = db.query(
        Laptop.brand,
        func.avg(Laptop.price).label('avg_price'),
        func.count(Laptop.id).label('count'),
        func.min(Laptop.price).label('min_price'),
        func.max(Laptop.price).label('max_price')
    ).group_by(Laptop.brand).all()

    return [
        {
            "brand": r.brand,
            "avg_price": round(r.avg_price, 2),
            "count": r.count,
            "min_price": r.min_price,
            "max_price": r.max_price
        }
        for r in results
    ]

@router.get("/price_dist", response_model=list[PriceDistributionItem])
async def get_price_distribution(
    bins: int = Query(10, ge=5, le=20, description="区间数量"),
    db: Session = Depends(get_db)
):
    """价格区间分布（直方图数据）"""
    # 获取价格最大最小值
    result = db.query(
        func.min(Laptop.price).label('min'),
        func.max(Laptop.price).label('max')
    ).first()

    min_price = result.min or 0
    max_price = result.max or 100000
    step = (max_price - min_price) / bins

    # 使用SQL CASE语句进行分桶（SQLite）
    distribution = []
    for i in range(bins):
        lower = min_price + i * step
        upper = min_price + (i + 1) * step
        if i == bins - 1:
            # 最后一个区间包含最大值
            count = db.query(Laptop).filter(
                Laptop.price >= lower,
                Laptop.price <= upper
            ).count()
        else:
            count = db.query(Laptop).filter(
                Laptop.price >= lower,
                Laptop.price < upper
            ).count()
        distribution.append({
            "range_label": f"{int(lower)}-{int(upper)}",
            "count": count
        })

    return distribution

@router.get("/trend", response_model=list[TrendItem])
async def get_price_trend(
    days: int = Query(30, ge=7, le=180, description="最近N天"),
    db: Session = Depends(get_db)
):
    """价格时序走势（按日聚合）"""
    cutoff_date = datetime.now().date() - timedelta(days=days)

    results = db.query(
        Laptop.crawl_date,
        func.avg(Laptop.price).label('avg_price'),
        func.count(Laptop.id).label('count')
    ).filter(
        Laptop.crawl_date >= cutoff_date
    ).group_by(
        Laptop.crawl_date
    ).order_by(
        Laptop.crawl_date
    ).all()

    return [
        {
            "date": r.crawl_date.strftime("%Y-%m-%d"),
            "avg_price": round(r.avg_price, 2),
            "count": r.count
        }
        for r in results
    ]

@router.get("/scatter", response_model=list[ScatterItem])
async def get_scatter_data(db: Session = Depends(get_db)):
    """价格-内存散点图数据（含品牌和评价）"""
    results = db.query(
        Laptop.brand,
        Laptop.price,
        Laptop.memory,
        Laptop.rating
    ).filter(
        Laptop.memory.isnot(None),
        Laptop.price > 0
    ).limit(500).all()

    return [
        {
            "brand": r.brand,
            "price": r.price,
            "memory": r.memory,
            "rating": r.rating
        }
        for r in results
    ]

@router.get("/ratio_rank", response_model=list[RatioRankItem])
async def get_ratio_rank(
    limit: int = Query(10, ge=5, le=20),
    db: Session = Depends(get_db)
):
    """
    性价比排名TOP N
    注：性价比指数需要先通过数据清洗/分析模块计算并存入数据库
    这里提供简化版本，实际可由P4分析模块计算后存入单独表
    """
    # 简化版：使用价格/内存作为性价比代理指标
    # 正式版应该调用P4的analysis模块结果
    results = db.query(
        Laptop.brand,
        Laptop.model,
        Laptop.price,
        Laptop.memory
    ).filter(
        Laptop.memory.isnot(None),
        Laptop.price > 0
    ).limit(limit).all()

    # 计算简化性价比 = 内存/价格*1000
    return [
        {
            "brand": r.brand,
            "model": r.model,
            "price": r.price,
            "performance_score": round(r.memory * 2, 2),  # 简化
            "value_for_money": round(r.memory / r.price * 1000, 4)
        }
        for r in results
    ]

from fastapi import APIRouter, Depends, Query, HTTPException
from sqlalchemy.orm import Session
from typing import Optional

from ..database import get_db
from ..models import Laptop
from ..schemas import LaptopResponse

router = APIRouter()

@router.get("/", response_model=list[LaptopResponse])
async def get_laptops(
    brand: Optional[str] = Query(None, description="品牌"),
    min_price: Optional[float] = Query(None, description="最低价"),
    max_price: Optional[float] = Query(None, description="最高价"),
    min_memory: Optional[int] = Query(None, description="最小内存GB"),
    limit: int = Query(100, ge=1, le=500),
    offset: int = Query(0, ge=0),
    db: Session = Depends(get_db)
):
    query = db.query(Laptop)
    if brand:
        query = query.filter(Laptop.brand == brand)
    if min_price:
        query = query.filter(Laptop.price >= min_price)
    if max_price:
        query = query.filter(Laptop.price <= max_price)
    if min_memory:
        query = query.filter(Laptop.memory >= min_memory)
    return query.offset(offset).limit(limit).all()

@router.get("/{laptop_id}", response_model=LaptopResponse)
async def get_laptop(laptop_id: int, db: Session = Depends(get_db)):
    laptop = db.query(Laptop).filter(Laptop.id == laptop_id).first()
    if not laptop:
        raise HTTPException(404, "笔记本不存在")
    return laptop

python
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from sqlalchemy import func
from datetime import datetime, timedelta

from ..database import get_db
from ..models import Laptop
from ..schemas import BrandAvgResponse, PriceDistItem, TrendItem, ScatterItem, RatioRankItem

router = APIRouter()

@router.get("/brand_avg", response_model=list[BrandAvgResponse])
async def get_brand_avg(db: Session = Depends(get_db)):
    results = db.query(
        Laptop.brand,
        func.avg(Laptop.price).label('avg_price'),
        func.count(Laptop.id).label('count')
    ).group_by(Laptop.brand).all()
    return [{"brand": r.brand, "avg_price": round(r.avg_price, 2), "count": r.count} for r in results]

@router.get("/price_dist", response_model=list[PriceDistItem])
async def get_price_dist(bins: int = Query(10, ge=5), db: Session = Depends(get_db)):
    # 简化实现：固定区间
    ranges = [(0, 3000), (3000, 5000), (5000, 7000), (7000, 10000), (10000, 15000), (15000, 30000)]
    result = []
    for low, high in ranges:
        count = db.query(Laptop).filter(Laptop.price >= low, Laptop.price < high).count()
        result.append({"range_label": f"{low}-{high}", "count": count})
    return result

@router.get("/trend", response_model=list[TrendItem])
async def get_trend(days: int = Query(30), db: Session = Depends(get_db)):
    cutoff = datetime.now().date() - timedelta(days=days)
    results = db.query(
        Laptop.crawl_date,
        func.avg(Laptop.price).label('avg_price')
    ).filter(Laptop.crawl_date >= cutoff).group_by(Laptop.crawl_date).order_by(Laptop.crawl_date).all()
    return [{"date": r.crawl_date.strftime("%Y-%m-%d"), "avg_price": round(r.avg_price, 2)} for r in results]

@router.get("/scatter", response_model=list[ScatterItem])
async def get_scatter(db: Session = Depends(get_db)):
    results = db.query(Laptop.brand, Laptop.price, Laptop.memory, Laptop.rating).filter(Laptop.memory.isnot(None)).limit(300).all()
    return [{"brand": r.brand, "price": r.price, "memory": r.memory, "rating": r.rating} for r in results]

@router.get("/ratio_rank", response_model=list[RatioRankItem])
async def get_ratio_rank(limit: int = Query(10), db: Session = Depends(get_db)):
    # 性价比 = 内存/价格*100（简化版，完整版由P4提供）
    results = db.query(Laptop.brand, Laptop.model, Laptop.price, Laptop.memory).filter(
        Laptop.memory.isnot(None), Laptop.price > 0
    ).limit(limit).all()
    return [{"brand": r.brand, "model": r.model, "price": r.price, "value_for_money": round(r.memory / r.price * 100, 4)} for r in results]

import sqlite3
import random
from datetime import datetime, timedelta

def init_test_data():
    conn = sqlite3.connect("data/laptops.db")
    cursor = conn.cursor()

    # 创建表
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS laptops (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            brand TEXT, model TEXT, price REAL,
            cpu_brand TEXT, cpu_model TEXT,
            memory INTEGER, storage INTEGER,
            gpu_model TEXT, rating REAL,
            review_count INTEGER, source TEXT, crawl_date DATE
        )
    """)

    # 清空旧数据
    cursor.execute("DELETE FROM laptops")

    brands = ["联想", "华硕", "戴尔", "惠普", "苹果", "华为", "小米", "宏碁"]
    models = ["ThinkPad", "拯救者", "灵越", "游匣", "MacBook", "MateBook", "RedmiBook", "暗影精灵"]
    cpu_brands = ["Intel", "AMD"]
    cpu_models = ["i5-12400H", "i7-12700H", "i9-12900H", "R5-6600H", "R7-6800H", "R9-6900HX"]
    gpu_models = ["RTX 3050", "RTX 3060", "RTX 3070", "RTX 4050", "RTX 4060", "集显"]

    for i in range(200):
        brand = random.choice(brands)
        price = random.randint(3000, 25000)
        memory = random.choice([8, 16, 32, 64])
        storage = random.choice([256, 512, 1024])
        rating = round(random.uniform(0.85, 0.99), 2)
        review_count = random.randint(100, 20000)
        days_ago = random.randint(1, 90)
        crawl_date = (datetime.now() - timedelta(days=days_ago)).strftime("%Y-%m-%d")

        cursor.execute("""
            INSERT INTO laptops (brand, model, price, cpu_brand, cpu_model, memory, storage, gpu_model, rating, review_count, source, crawl_date)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            brand,
            f"{brand} {random.choice(models)} {i}",
            price,
            random.choice(cpu_brands),
            random.choice(cpu_models),
            memory,
            storage,
            random.choice(gpu_models),
            rating,
            review_count,
            "测试数据",
            crawl_date
        ))

    conn.commit()
    conn.close()
